```tsx
import { lazy, Suspense, useEffect, useState } from "react";
import {
  CalendarClock,
  ClipboardList,
  PillBottle,
  TriangleAlert,
} from "lucide-react";

import { Tabs, TabsContent, TabsList, TabsTrigger } from "@/components/ui/tabs";
import { AddMedicineForm, CabinetList } from "@/features/AddMedicineForm";
import type { Medicine, ScheduledDose } from "@/types/medicine";
import { generateScheduleForMedicine } from "@/lib/schedule";
import { WelcomeView } from "@/features/WelcomeView";

// Route-level code splitting: only the default "Add medicine" tab ships in the
// initial chunk. The other three views load on demand, avoiding an eager
// waterfall of unused feature code on first paint.
const ScheduleView = lazy(() =>
  import("@/features/ScheduleView").then((mod) => ({
    default: mod.ScheduleView,
  })),
);
const InventoryLedger = lazy(() =>
  import("@/features/InventoryLedger").then((mod) => ({
    default: mod.InventoryLedger,
  })),
);
const ExpirationBoard = lazy(() =>
  import("@/features/ExpirationBoard").then((mod) => ({
    default: mod.ExpirationBoard,
  })),
);

const MEDICINES_STORAGE_KEY = "medidose_medicines";
const DOSES_STORAGE_KEY = "medidose_doses";
const ACTIVE_TAB_STORAGE_KEY = "medidose_active_tab";

/** Safely reads and parses a JSON array from localStorage, falling back to [] on any problem. */
function loadFromStorage<T>(key: string): T[] {
  if (typeof window === "undefined") return [];
  try {
    const raw = window.localStorage.getItem(key);
    if (!raw) return [];
    const parsed = JSON.parse(raw);
    return Array.isArray(parsed) ? (parsed as T[]) : [];
  } catch {
    return [];
  }
}

/** Lightweight placeholder shown while a lazily-loaded tab view streams in. */
function ViewSkeleton() {
  return (
    <div className="flex flex-col gap-3" aria-hidden="true">
      <div className="h-20 animate-pulse rounded-2xl border border-border/50 bg-card" />
      <div className="h-20 animate-pulse rounded-2xl border border-border/50 bg-card" />
      <div className="h-20 animate-pulse rounded-2xl border border-border/50 bg-card" />
    </div>
  );
}

function App() {
  // State is initialized from localStorage (if present) and persisted on every change.
  
  const [medicines, setMedicines] = useState<Medicine[]>(() =>
    loadFromStorage<Medicine>(MEDICINES_STORAGE_KEY),
  );
  const [doses, setDoses] = useState<ScheduledDose[]>(() =>
    loadFromStorage<ScheduledDose>(DOSES_STORAGE_KEY),
  );

  const [activeTab, setActiveTab] = useState<string>(() => {
    if (typeof window === "undefined") return "welcome";
    const savedTab = window.localStorage.getItem(ACTIVE_TAB_STORAGE_KEY);
    if (savedTab) return savedTab;
    return medicines.length === 0 ? "welcome" : "add";
  });

  useEffect(() => {
    try {
      window.localStorage.setItem(ACTIVE_TAB_STORAGE_KEY, activeTab);
    } catch {
      // Storage unavailable fallback
    }
  }, [activeTab]);

  useEffect(() => {
    try {
      window.localStorage.setItem(
        MEDICINES_STORAGE_KEY,
        JSON.stringify(medicines),
      );
    } catch {
      // Storage may be unavailable (private browsing, quota exceeded, etc.) — fail silently.
    }
  }, [medicines]);

  useEffect(() => {
    try {
      window.localStorage.setItem(DOSES_STORAGE_KEY, JSON.stringify(doses));
    } catch {
      // Storage may be unavailable (private browsing, quota exceeded, etc.) — fail silently.
    }
  }, [doses]);

  function addMedicine(medicine: Medicine) {
    setMedicines((prev) => [...prev, medicine]);
    setDoses((prev) => [...prev, ...generateScheduleForMedicine(medicine)]);
  }

  /** Marks a scheduled dose taken/untaken, decrementing or restoring cabinet stock to match. */
  function toggleDose(doseId: string) {
    const dose = doses.find((d) => d.id === doseId);
    if (!dose) return;
    const medicine = medicines.find((m) => m.id === dose.medicineId);
    if (!medicine) return;

    // Block logging a dose when there's no stock left to log it against.
    if (!dose.taken && medicine.stock <= 0) return;

    setMedicines((prev) =>
      prev.map((m) =>
        m.id === medicine.id
          ? { ...m, stock: dose.taken ? m.stock + 1 : Math.max(0, m.stock - 1) }
          : m,
      ),
    );
    setDoses((prev) =>
      prev.map((d) => (d.id === doseId ? { ...d, taken: !d.taken } : d)),
    );
  }

  /** Adds stock to an existing medicine, e.g. after buying a refill. */
  function restockMedicine(medicineId: string, additionalStock: number) {
    if (!Number.isFinite(additionalStock) || additionalStock <= 0) return;
    setMedicines((prev) =>
      prev.map((m) =>
        m.id === medicineId ? { ...m, stock: m.stock + additionalStock } : m,
      ),
    );
  }

  /** Removes a medicine and its schedule entirely, e.g. to fix an accidental entry or typo. */
  function deleteMedicine(medicineId: string) {
    setMedicines((prev) => prev.filter((m) => m.id !== medicineId));
    setDoses((prev) => prev.filter((d) => d.medicineId !== medicineId));
  }

  return (
    <div className="min-h-[100dvh] bg-background font-['Geist',_ui-sans-serif,_system-ui,_sans-serif]">
      <header className="border-b border-border/60 bg-card">
        <div className="mx-auto flex max-w-4xl items-center gap-3 px-4 py-5 md:px-6">
          <div className="flex size-9 items-center justify-center rounded-2xl bg-sage-soft text-sage">
            <PillBottle className="size-5" />
          </div>
          <div>
            <h1 className="font-display text-xl font-medium leading-none tracking-tight">
              MediDose
            </h1>
            <p className="mt-1 text-sm text-muted-foreground">
              The home medicine cabinet, kept in order.
            </p>
          </div>
        </div>
      </header>

      <main className="container mx-auto max-w-4xl px-4 py-8 md:px-6">
        {activeTab === "welcome" && medicines.length === 0 ? (
          <WelcomeView onStartAdding={() => setActiveTab("add")} />
        ) : (
          <Tabs value={activeTab} onValueChange={setActiveTab}>
            <TabsList>
              <TabsTrigger value="add">
                <PillBottle /> Add medicine
              </TabsTrigger>
              <TabsTrigger value="schedule">
                <CalendarClock /> Today's schedule
              </TabsTrigger>
              <TabsTrigger value="inventory">
                <ClipboardList /> Inventory ledger
              </TabsTrigger>
              <TabsTrigger value="alerts">
                <TriangleAlert /> Expiration alerts
              </TabsTrigger>
            </TabsList>

            <TabsContent value="add" className="flex flex-col gap-6">
              <AddMedicineForm onAdd={addMedicine} />
              <CabinetList medicines={medicines} onDelete={deleteMedicine} />
            </TabsContent>

            <TabsContent value="schedule">
              <Suspense fallback={<ViewSkeleton />}>
                <ScheduleView
                  medicines={medicines}
                  doses={doses}
                  onToggleDose={toggleDose}
                />
              </Suspense>
            </TabsContent>

            <TabsContent value="inventory">
              <Suspense fallback={<ViewSkeleton />}>
                <InventoryLedger
                  medicines={medicines}
                  onRestock={restockMedicine}
                  onDelete={deleteMedicine}
                />
              </Suspense>
            </TabsContent>

            <TabsContent value="alerts">
              <Suspense fallback={<ViewSkeleton />}>
                <ExpirationBoard
                  medicines={medicines}
                  onDispose={deleteMedicine}
                  onDelete={deleteMedicine}
                />
              </Suspense>
            </TabsContent>
          </Tabs>
        )}
      </main>
    </div>
  );
}

export default App;

```

```tsx
import { type ReactNode } from "react"
import { Trash2 } from "lucide-react"
import { Button } from "@/components/ui/button"
import {
  AlertDialog,
  AlertDialogAction,
  AlertDialogCancel,
  AlertDialogContent,
  AlertDialogDescription,
  AlertDialogFooter,
  AlertDialogHeader,
  AlertDialogTitle,
  AlertDialogTrigger,
} from "@/components/ui/alert-dialog"

interface DeleteMedicineDialogProps {
  medicineName: string
  onConfirm: () => void
  /** Optional custom trigger. Defaults to a ghost button with a Trash icon. */
  trigger?: ReactNode
  /** Icon color override. Options: 'muted' | 'destructive'. Defaults to 'destructive'. */
  iconVariant?: "muted" | "destructive"
}

export function DeleteMedicineDialog({
  medicineName,
  onConfirm,
  trigger,
  iconVariant = "destructive",
}: DeleteMedicineDialogProps) {
  const iconColorClass =
    iconVariant === "muted" ? "text-muted-foreground" : "text-destructive"

  return (
    <AlertDialog>
      <AlertDialogTrigger asChild>
        {trigger ?? (
          <Button
            size="sm"
            variant="ghost"
            aria-label={`Remove ${medicineName}`}
          >
            <Trash2 className={`size-4 ${iconColorClass}`} />
          </Button>
        )}
      </AlertDialogTrigger>
      <AlertDialogContent>
        <AlertDialogHeader>
          <AlertDialogTitle>Are you sure?</AlertDialogTitle>
          <AlertDialogDescription>
            Remove {medicineName} from the cabinet? This can&apos;t be undone.
          </AlertDialogDescription>
        </AlertDialogHeader>
        <AlertDialogFooter>
          <AlertDialogCancel>Cancel</AlertDialogCancel>
          <AlertDialogAction
            onClick={onConfirm}
            variant={"destructive"}
          >
            Remove
          </AlertDialogAction>
        </AlertDialogFooter>
      </AlertDialogContent>
    </AlertDialog>
  )
}
```

```tsx
import * as React from "react"
import { AlertDialog as AlertDialogPrimitive } from "radix-ui"

import { cn } from "@/lib/utils"
import { Button } from "@/components/ui/button"

function AlertDialog({
  ...props
}: React.ComponentProps<typeof AlertDialogPrimitive.Root>) {
  return <AlertDialogPrimitive.Root data-slot="alert-dialog" {...props} />
}

function AlertDialogTrigger({
  ...props
}: React.ComponentProps<typeof AlertDialogPrimitive.Trigger>) {
  return (
    <AlertDialogPrimitive.Trigger data-slot="alert-dialog-trigger" {...props} />
  )
}

function AlertDialogPortal({
  ...props
}: React.ComponentProps<typeof AlertDialogPrimitive.Portal>) {
  return (
    <AlertDialogPrimitive.Portal data-slot="alert-dialog-portal" {...props} />
  )
}

function AlertDialogOverlay({
  className,
  ...props
}: React.ComponentProps<typeof AlertDialogPrimitive.Overlay>) {
  return (
    <AlertDialogPrimitive.Overlay
      data-slot="alert-dialog-overlay"
      className={cn(
        // Ink-tinted scrim (reuses the existing `foreground` token) instead of
        // raw black, plus a touch of blur for materiality.
        "fixed inset-0 z-50 bg-foreground/40 backdrop-blur-[2px] data-[state=closed]:animate-out data-[state=closed]:fade-out-0 data-[state=open]:animate-in data-[state=open]:fade-in-0",
        className
      )}
      {...props}
    />
  )
}

function AlertDialogContent({
  className,
  size = "default",
  ...props
}: React.ComponentProps<typeof AlertDialogPrimitive.Content> & {
  size?: "default" | "sm"
}) {
  return (
    <AlertDialogPortal>
      <AlertDialogOverlay />
      <AlertDialogPrimitive.Content
        data-slot="alert-dialog-content"
        data-size={size}
        className={cn(
          "group/alert-dialog-content fixed top-[50%] left-[50%] z-50 grid w-full max-w-[calc(100%-2rem)] translate-x-[-50%] translate-y-[-50%] gap-4 rounded-3xl border border-border/50 bg-background p-6 shadow-lg shadow-black/[0.06] duration-200 data-[size=sm]:max-w-xs data-[state=closed]:animate-out data-[state=closed]:fade-out-0 data-[state=closed]:zoom-out-95 data-[state=open]:animate-in data-[state=open]:fade-in-0 data-[state=open]:zoom-in-95 data-[size=default]:sm:max-w-lg",
```

```tsx
  onAdd: (medicine: Medicine) => void;
}

export function AddMedicineForm({ onAdd }: AddMedicineFormProps) {
  const [values, setValues] = useState<MedicineFormValues>(emptyForm);
  const [errors, setErrors] = useState<MedicineFormErrors>({});

  function setField<K extends keyof MedicineFormValues>(
    key: K,
    value: MedicineFormValues[K],
  ) {
    setValues((prev) => ({ ...prev, [key]: value }));
  }

  function handleFrequencyChange(newFreqStr: string) {
    const newFreq = Number(newFreqStr);
    setField("frequencyPerDay", newFreqStr);

    if (values.scheduleMode === "custom") {
      const currentTimes = [...values.customTimes];
      if (currentTimes.length < newFreq) {
        while (currentTimes.length < newFreq) {
          currentTimes.push("12:00");
        }
      } else if (currentTimes.length > newFreq) {
        currentTimes.splice(newFreq);
      }
      setField("customTimes", currentTimes);
    }
  }

  function handleModeChange(newMode: ScheduleMode) {
    setField("scheduleMode", newMode);
    if (newMode === "custom") {
      const freq = Number(values.frequencyPerDay) || 1;
      const initialTimes = Array.from({ length: freq }, (_, i) => {
        return values.customTimes[i] || values.firstDoseTime || "08:00";
      });
      setField("customTimes", initialTimes);
    }
  }

    const nextTimes = [...values.customTimes];
    nextTimes[index] = newTime;
    setField("customTimes", nextTimes);

    const nextErrors = validateMedicineForm(values);
    setErrors(nextErrors);

      id: crypto.randomUUID(),
      name: values.name.trim(),
      dosageAmount: Number(values.dosageAmount),
      dosageUnit: values.dosageUnit,
      stock: Number(values.stock),
      expirationDate: values.expirationDate,
      frequencyPerDay: Number(values.frequencyPerDay),
      patientName: values.patientName.trim(),
      firstDoseTime: values.firstDoseTime || "08:00",
      treatmentEndDate: values.treatmentEndDate || undefined,
      scheduleMode: values.scheduleMode,

    };

                  </SelectItem>
```

```tsx
interface ScheduleViewProps {
  medicines: Medicine[];

function computeVisibleDoseIds(
  medicines: Medicine[],
  doses: ScheduledDose[],
): Set<string> {
  const visible = new Set<string>();
  const dosesByMedicine = new Map<string, ScheduledDose[]>();

  for (const dose of doses) {
    const list = dosesByMedicine.get(dose.medicineId);
    if (list) {
      list.push(dose);
    } else {
      dosesByMedicine.set(dose.medicineId, [dose]);
    }
  }

  for (const medicine of medicines) {
    const medicineDoses = (dosesByMedicine.get(medicine.id) ?? [])
      .slice()
      .sort((a, b) => (a.date + a.time).localeCompare(b.date + b.time));

    let remainingStock = medicine.stock;
    for (const dose of medicineDoses) {
      if (dose.taken) {
        visible.add(dose.id);
        continue;
      }
      if (remainingStock > 0) {
        visible.add(dose.id);
        remainingStock -= 1;
      }
    }
  }

  return visible;
}

export function ScheduleView({
  medicines,
  doses,
  onToggleDose,
}: ScheduleViewProps) {
  const [selectedPatient, setSelectedPatient] = useState<string>("all");
  const todayIso = toISODate(today());

  const medicineById = useMemo(
    () => new Map(medicines.map((m) => [m.id, m])),
    [medicines],
  );

  const visibleDoseIds = useMemo(
    () => computeVisibleDoseIds(medicines, doses),
    [medicines, doses],
  );

  // Extract unique patient names for filter tabs
  const patients = useMemo(() => {
    const set = new Set(medicines.map((m) => m.patientName));
    return Array.from(set);
  }, [medicines]);

  // Group today's doses by time slot
  const todayTimeGroups = useMemo(() => {
    const timeMap = new Map<string, ScheduledDose[]>();

    const filteredDoses = doses.filter((d) => {
      if (d.date !== todayIso || !visibleDoseIds.has(d.id)) return false;
      if (selectedPatient === "all") return true;
      const med = medicineById.get(d.medicineId);
      return med?.patientName === selectedPatient;
    });

    for (const dose of filteredDoses) {
      const group = timeMap.get(dose.time);
      if (group) {
        group.push(dose);
      } else {
        timeMap.set(dose.time, [dose]);
      }
    }

    // Sort time slots chronologically
    return Array.from(timeMap.entries()).sort(([timeA], [timeB]) =>
```

```ts
/** Returns today at midnight, local time, as a Date. */
export function today(): Date {
  const d = new Date()
  d.setHours(0, 0, 0, 0)
  return d
}

/** Parses a "YYYY-MM-DD" string into a local-midnight Date. */
export function parseISODate(iso: string): Date {
  const [year, month, day] = iso.split("-").map(Number)
  return new Date(year, month - 1, day)
}

/** Formats a Date as "YYYY-MM-DD". */
export function toISODate(date: Date): string {
  const year = date.getFullYear()
  const month = String(date.getMonth() + 1).padStart(2, "0")
  const day = String(date.getDate()).padStart(2, "0")
  return `${year}-${month}-${day}`
}

/** Whole number of days from today until the given ISO date (negative if past). */
export function daysUntil(iso: string): number {
  const target = parseISODate(iso)
  const diffMs = target.getTime() - today().getTime()
  return Math.round(diffMs / (1000 * 60 * 60 * 24))
}

/** True if the given ISO date is strictly before today. */
export function isPastDate(iso: string): boolean {
  return daysUntil(iso) < 0
}

/** Human-friendly relative label, e.g. "Expires in 12 days" / "Expired 3 days ago". */
export function formatDaysRemaining(iso: string): string {
  const days = daysUntil(iso)
  if (days < 0) return `Expired ${Math.abs(days)} day${Math.abs(days) === 1 ? "" : "s"} ago`
  if (days === 0) return "Expires today"
  return `${days} day${days === 1 ? "" : "s"} left`
}
```

```ts
// src/lib/schedule.ts
import type { Medicine, ScheduledDose } from "@/types/medicine"
import { today, toISODate } from "@/lib/dates"

const SCHEDULE_DAYS = 7
/** Default first dose of the day when a medicine has no custom start time set. */
const DEFAULT_FIRST_DOSE_TIME = "08:00"
/** Waking hours cutoff for Waking Hours mode (10:00 PM). */
const WAKING_HOURS_END = "22:00"

/** Converts "HH:MM" 24hr string into total minutes from midnight. */
function timeToMinutes(time: string): number {
  const [hours, minutes] = time.split(":").map(Number)
  return hours * 60 + minutes
}

/** Formats total minutes from midnight into a "HH:MM" 24hr string. */
function minutesToTime(totalMinutes: number): string {
  const wrapped = ((totalMinutes % 1440) + 1440) % 1440
  const hours = Math.floor(wrapped / 60)
  const mins = Math.round(wrapped % 60)
  return `${String(hours).padStart(2, "0")}:${String(mins).padStart(2, "0")}`
}

/**
 * Calculates schedule time slots based on the medicine's schedule mode:
 * 1. "waking_hours" (Default): Spreads doses between firstDoseTime and 10:00 PM.
 * 2. "strict_interval": Calculates exact 24/N intervals (e.g., 4x/day -> every 6 hours).
 * 3. "custom": Returns the explicitly specified custom times array.
 */
export function timeSlotsForFrequency(
  frequency: number,
  firstDoseTime: string = DEFAULT_FIRST_DOSE_TIME,
  scheduleMode: "waking_hours" | "strict_interval" | "custom" = "waking_hours",
  customTimes: string[] = []
): string[] {
  if (scheduleMode === "custom" && customTimes.length > 0) {
    return [...customTimes].sort()
  }

  if (frequency <= 1) {
    return [firstDoseTime]
  }

  const startMins = timeToMinutes(firstDoseTime)

  if (scheduleMode === "strict_interval") {
    const intervalMins = Math.round((24 / frequency) * 60)
    const slots: string[] = []
    for (let i = 0; i < frequency; i++) {
      slots.push(minutesToTime(startMins + i * intervalMins))
    }
    return slots.sort()
  }

  // Default: "waking_hours" mode
  const endMins = timeToMinutes(WAKING_HOURS_END)
  let windowMins = endMins - startMins

  // Fallback to strict 24h interval if starting at or after waking cutoff
  if (windowMins <= 0) {
    const intervalMins = Math.round((24 / frequency) * 60)
    const slots: string[] = []
    for (let i = 0; i < frequency; i++) {
      slots.push(minutesToTime(startMins + i * intervalMins))
    }
    return slots.sort()
  }

  const step = windowMins / (frequency - 1)
  const slots: string[] = []
  for (let i = 0; i < frequency; i++) {
    slots.push(minutesToTime(startMins + Math.round(i * step)))
  }
  return slots.sort()
}

/** Builds the full 7-day set of scheduled doses for one medicine, starting today. */
export function generateScheduleForMedicine(medicine: Medicine): ScheduledDose[] {
  const slots = timeSlotsForFrequency(
    medicine.frequencyPerDay,
    medicine.firstDoseTime,
    medicine.scheduleMode,
    medicine.customTimes
  )
  const doses: ScheduledDose[] = []
  const start = today()

  for (let dayOffset = 0; dayOffset < SCHEDULE_DAYS; dayOffset++) {
    const date = new Date(start)
    date.setDate(date.getDate() + dayOffset)
    const iso = toISODate(date)

    // Stop generating schedule slots past the treatment end date
    if (medicine.treatmentEndDate && iso > medicine.treatmentEndDate) {
      break
    }

    slots.forEach((time, slotIndex) => {
      doses.push({
        id: `${medicine.id}-${iso}-${slotIndex}`,
        medicineId: medicine.id,
        date: iso,
        time,
        taken: false,
      })
    })
  }

  return doses
}

/** Formats a "HH:MM" 24hr time string as 12hr with AM/PM, e.g. "08:00" -> "8:00 AM". */
export function formatTime12h(time: string): string {
  const [hours, minutes] = time.split(":").map(Number)
  const period = hours >= 12 ? "PM" : "AM"
  const hour12 = hours % 12 === 0 ? 12 : hours % 12
  return `${hour12}:${String(minutes).padStart(2, "0")} ${period}`
}
```

```ts
// src/lib/validation.ts
import { MAX_FREQUENCY, MIN_FREQUENCY, type DoseUnit, type ScheduleMode } from "@/types/medicine";
import { isPastDate } from "@/lib/dates";

export interface MedicineFormValues {
  name: string;
  dosageAmount: string;
  dosageUnit: DoseUnit;
  stock: string;
  expirationDate: string;
  frequencyPerDay: string;
  patientName: string;
  firstDoseTime: string;
  treatmentEndDate: string;
  scheduleMode: ScheduleMode;
  customTimes: string[];
}

export type MedicineFormErrors = Partial<
  Record<keyof MedicineFormValues, string>
>;

/**
 * Regex for valid names:
 * Allows letters (Unicode/accented characters), numbers, spaces, hyphens, apostrophes, and periods.
 * Disallows symbols and special characters like @, #, $, %, <, >, /, etc.
 */
const VALID_NAME_REGEX = /^[a-zA-Z0-9\s'\-\.]+$/;

/** Regex to check if a string contains ONLY numbers and spaces/punctuation (no letters) */
const NUMBERS_ONLY_REGEX = /^[\d\s'\-\.]+$/;

/** Validates a new-medicine form. Returns a map of field -> error message. */
export function validateMedicineForm(
  values: MedicineFormValues,
): MedicineFormErrors {
  const errors: MedicineFormErrors = {};

  // 1. Medicine Name Validation
  const trimmedName = values.name.trim();
  if (!trimmedName) {
    errors.name = "Enter the medicine's brand or generic name.";
  } else if (!VALID_NAME_REGEX.test(trimmedName)) {
    errors.name = "Medicine name can only contain letters, numbers, spaces, hyphens, periods, and apostrophes.";
  } else if (NUMBERS_ONLY_REGEX.test(trimmedName)) {
    errors.name = "Medicine name cannot contain only numbers. Please include letters.";
  }

  // 2. Patient Name Validation
  const trimmedPatient = values.patientName.trim();
  if (!trimmedPatient) {
    errors.patientName = "Enter who this medicine is for.";
  } else if (!VALID_NAME_REGEX.test(trimmedPatient)) {
    errors.patientName = "Patient name can only contain letters, numbers, spaces, hyphens, periods, and apostrophes.";
  } else if (NUMBERS_ONLY_REGEX.test(trimmedPatient)) {
    errors.patientName = "Patient name cannot contain only numbers. Please include letters.";
  }

  const dosageAmount = Number(values.dosageAmount);
  if (!values.dosageAmount || Number.isNaN(dosageAmount) || dosageAmount <= 0) {
    errors.dosageAmount = "Enter a dosage amount greater than 0.";
  }

  const stock = Number(values.stock);
  if (
    values.stock === "" ||
    Number.isNaN(stock) ||
    stock < 0 ||
    !Number.isInteger(stock)
  ) {
    errors.stock = "Enter a whole number of doses in stock (0 or more).";
  }

  if (!values.expirationDate) {
    errors.expirationDate = "Choose an expiration date.";
  } else if (isPastDate(values.expirationDate)) {
    errors.expirationDate = "Expiration date can't be in the past.";
  }

  const frequency = Number(values.frequencyPerDay);
  if (
    !values.frequencyPerDay ||
    Number.isNaN(frequency) ||
    !Number.isInteger(frequency) ||
    frequency < MIN_FREQUENCY ||
    frequency > MAX_FREQUENCY
  ) {
    errors.frequencyPerDay = `Frequency must be a whole number between ${MIN_FREQUENCY} and ${MAX_FREQUENCY}.`;
  }

  if (values.scheduleMode !== "custom") {
    if (
      values.firstDoseTime &&
      !/^([01]\d|2[0-3]):[0-5]\d$/.test(values.firstDoseTime)
    ) {
      errors.firstDoseTime = "Enter a valid time for the first dose.";
    }
  }

  if (values.scheduleMode === "custom") {
    if (!values.customTimes || values.customTimes.length === 0) {
      errors.customTimes = "Add at least one time for custom schedules.";
    } else {
      const invalidTime = values.customTimes.some(
        (t) => !/^([01]\d|2[0-3]):[0-5]\d$/.test(t)
      );
      if (invalidTime) {
        errors.customTimes = "All custom times must be valid HH:MM formats.";
      }
    }
  }

  if (values.treatmentEndDate) {
    if (isPastDate(values.treatmentEndDate)) {
      errors.treatmentEndDate = "Treatment end date can't be in the past.";
    }
  }

  return errors;
}

export function hasErrors(errors: MedicineFormErrors): boolean {
  return Object.keys(errors).length > 0;
}
```
